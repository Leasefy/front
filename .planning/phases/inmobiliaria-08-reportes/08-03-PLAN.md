```typescript
import type {
  CarteraReport,
  ComisionesAgenteReport,
  VencimientosReport,
  FlujoCajaReport,
} from '@/lib/types/inmobiliaria';
import { formatCurrency } from '@/lib/format';

// Generic CSV generator
function generateCSV(headers: string[], rows: string[][]): string {
  const headerRow = headers.join(',');
  const dataRows = rows.map((row) =>
    row.map((cell) => `"${String(cell).replace(/"/g, '""')}"`).join(',')
  );
  return [headerRow, ...dataRows].join('\n');
}

function downloadCSV(content: string, filename: string): void {
  const blob = new Blob(['\ufeff' + content], { type: 'text/csv;charset=utf-8;' });
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  link.download = filename;
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
}

export function exportCarteraEdades(data: CarteraReport): void {
  const headers = [
    'Propiedad',
    'Dirección',
    'Inquilino',
    'Teléfono',
    'Propietario',
    'Agente',
    'Mes',
    'Monto Total',
    'Pagado',
    'Pendiente',
    'Días de Mora',
    'Bucket',
  ];

  const rows = data.items.map((item) => [
    item.propertyTitle,
    item.propertyAddress,
    item.tenantName,
    item.tenantPhone,
    item.propietarioName,
    item.agenteName,
    item.month,
    formatCurrency(item.totalAmount),
    formatCurrency(item.paidAmount),
    formatCurrency(item.pendingAmount),
    item.daysLate.toString(),
    item.bucket,
  ]);

  const csv = generateCSV(headers, rows);
  downloadCSV(csv, `cartera-edades-${new Date().toISOString().split('T')[0]}.csv`);
}

export function exportComisionesAgente(data: ComisionesAgenteReport): void {
  const headers = [
    'Agente',
    'Cierres',
    'Comisión Total',
    'Promedio por Cierre',
    'Mejor Propiedad',
  ];

  const rows = data.items.map((item) => [
    item.agenteName,
    item.closedDeals.toString(),
    formatCurrency(item.totalCommission),
    formatCurrency(item.avgCommission),
    item.topProperty || '-',
  ]);

  const csv = generateCSV(headers, rows);
  downloadCSV(csv, `comisiones-agente-${new Date().toISOString().split('T')[0]}.csv`);
}

export function exportVencimientos(data: VencimientosReport): void {
  const headers = [
    'Propiedad',
    'Inquilino',
    'Propietario',
    'Fecha Vencimiento',
    'Días para Vencer',
    'Estado Renovación',
  ];

  const rows = data.items.map((item) => [
    item.propertyTitle,
    item.tenantName,
    item.propietarioName,
    item.leaseEnd,
    item.daysUntilExpiry.toString(),
    item.renewalStatus,
  ]);

  const csv = generateCSV(headers, rows);
  downloadCSV(csv, `vencimientos-${new Date().toISOString().split('T')[0]}.csv`);
}

export function exportFlujoCaja(data: FlujoCajaReport): void {
  const headers = [
    'Mes',
    'Ingresos',
    'Dispersiones',
    'Comisiones',
    'Balance',
  ];

  const rows = data.months.map((month) => [
    month.month,
    formatCurrency(month.ingresos),
    formatCurrency(month.dispersiones),
    formatCurrency(month.comisiones),
    formatCurrency(month.balance),
  ]);

  // Add summary row
  rows.push([
    'TOTAL',
    formatCurrency(data.summary.totalIngresos),
    formatCurrency(data.summary.totalDispersiones),
    formatCurrency(data.summary.totalComisiones),
    formatCurrency(data.summary.netBalance),
  ]);

  const csv = generateCSV(headers, rows);
  downloadCSV(csv, `flujo-caja-${new Date().toISOString().split('T')[0]}.csv`);
}
```
