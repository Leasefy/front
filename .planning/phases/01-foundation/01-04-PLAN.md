<task type="auto">
  <name>Task 2: Create seed script</name>
  <files>prisma/seed.ts, package.json</files>
  <action>
    Create the Prisma seed script:

    ```typescript
    // prisma/seed.ts
    import { db } from '../src/lib/db'
    import { landlords, tenants, properties, applications } from '../src/lib/seed-data'

    async function main() {
      console.log('🌱 Seeding database...')

      // Clear existing data (in correct order for FK constraints)
      await db.candidateNote.deleteMany()
      await db.applicationEvent.deleteMany()
      await db.riskScoreResult.deleteMany()
      await db.application.deleteMany()
      await db.propertyImage.deleteMany()
      await db.property.deleteMany()
      await db.user.deleteMany()

      console.log('✓ Cleared existing data')

      // Create users (landlords and tenants)
      await db.user.createMany({ data: [...landlords, ...tenants] })
      console.log(`✓ Created ${landlords.length} landlords and ${tenants.length} tenants`)

      // Create properties with images
      for (const property of properties) {
        const { images, ...propertyData } = property
        await db.property.create({
          data: {
            ...propertyData,
            images: {
              create: images
            }
          }
        })
      }
      console.log(`✓ Created ${properties.length} properties`)

      // Create applications
      await db.application.createMany({ data: applications })
      console.log(`✓ Created ${applications.length} applications`)

      console.log('🎉 Seeding complete!')
    }

    main()
      .catch((e) => {
        console.error('Seeding error:', e)
        process.exit(1)
      })
      .finally(async () => {
        await db.$disconnect()
      })
    ```

    Update package.json to configure Prisma seeding:

    ```json
    {
      "prisma": {
        "seed": "npx tsx prisma/seed.ts"
      },
      "scripts": {
        "db:seed": "npx prisma db seed",
        "db:reset": "npx prisma db push --force-reset && npm run db:seed"
      }
    }
    ```

    Install tsx for TypeScript execution:
    ```bash
    npm install -D tsx
    ```
  </action>
  <verify>npm run db:seed executes without errors</verify>
  <done>Seed script creates all demo data for MVP testing</done>
