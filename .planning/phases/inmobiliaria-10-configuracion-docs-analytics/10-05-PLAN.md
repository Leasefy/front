import type { ActaEntrega, InventoryItem, ItemCondition, RoomType } from '@/lib/types/inmobiliaria';

function generateInventoryItems(rooms: RoomType[]): InventoryItem[] {
  const items: InventoryItem[] = [];
  const conditions: ItemCondition[] = ['excelente', 'bueno', 'regular', 'malo'];

  rooms.forEach((room, roomIndex) => {
    const roomItems = COMMON_ITEMS_BY_ROOM[room] || [];
    roomItems.forEach((itemName, itemIndex) => {
      const condition = conditions[(roomIndex + itemIndex) % conditions.length];
      items.push({
        id: `item-${room}-${itemIndex}`,
        room,
        name: itemName,
        quantity: 1,
        condition,
        hasDefects: condition === 'malo' || condition === 'regular',
        defectDescription: condition === 'malo' ? 'Requiere reparacion' : undefined,
      });
    });
  });

  return items;
}
