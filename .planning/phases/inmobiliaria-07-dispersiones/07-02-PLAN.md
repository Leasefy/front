  <action>
Create PDF generation utility for extracto:

```typescript
import jsPDF from 'jspdf';
import type { ExtractoPropietario } from '@/lib/types/inmobiliaria';
import { formatCurrency } from '@/lib/format';

export function generateExtractoPropietarioPDF(extracto: ExtractoPropietario): jsPDF {
  const doc = new jsPDF();

  // Header with inmobiliaria info
  doc.setFontSize(18);
  doc.text('EXTRACTO DEL PROPIETARIO', 105, 20, { align: 'center' });

  // Period
  doc.setFontSize(12);
  doc.text(`Período: ${formatMonth(extracto.month)}`, 105, 30, { align: 'center' });

  // Propietario info section
  doc.setFontSize(10);
  doc.text(`Propietario: ${extracto.propietarioName}`, 20, 45);
  doc.text(`Generado: ${formatDate(extracto.generatedAt)}`, 20, 52);

  // Properties table header
  let y = 65;
  doc.setFillColor(240, 240, 240);
  doc.rect(20, y, 170, 8, 'F');
  doc.text('Propiedad', 22, y + 5);
  doc.text('Recaudado', 100, y + 5);
  doc.text('Comisión', 130, y + 5);
  doc.text('Neto', 160, y + 5);

  // Properties rows
  y += 10;
  extracto.properties.forEach((prop) => {
    doc.text(prop.propertyTitle.substring(0, 30), 22, y + 5);
    doc.text(formatCurrency(prop.totalCollected), 100, y + 5);
    doc.text(formatCurrency(prop.commissionAmount), 130, y + 5);
    doc.text(formatCurrency(prop.netAmount), 160, y + 5);
    y += 8;
  });

  // Summary
  y += 10;
  doc.setFillColor(230, 250, 230);
  doc.rect(20, y, 170, 20, 'F');
  doc.setFontSize(12);
  doc.text(`Total Neto a Recibir: ${formatCurrency(extracto.summary.netToPropietario)}`, 22, y + 12);

  // Footer
  doc.setFontSize(8);
  doc.text('Arriendos Premium S.A.S. - NIT 901234567-8', 105, 280, { align: 'center' });

  return doc;
}

export function downloadExtractoPDF(extracto: ExtractoPropietario): void {
  const doc = generateExtractoPropietarioPDF(extracto);
  doc.save(`extracto-${extracto.propietarioName.replace(/\s/g, '-')}-${extracto.month}.pdf`);
}
```
