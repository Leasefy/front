```typescript
const WISHLIST_KEY = 'arriendo-facil-wishlist';

export function useWishlist() {
  const [wishlistedIds, setWishlistedIds] = useState<string[]>([]);

  // Load from localStorage on mount
  useEffect(() => {
    const stored = localStorage.getItem(WISHLIST_KEY);
    if (stored) {
      setWishlistedIds(JSON.parse(stored));
    }
  }, []);

  // Save to localStorage on change
  useEffect(() => {
    localStorage.setItem(WISHLIST_KEY, JSON.stringify(wishlistedIds));
  }, [wishlistedIds]);

  const toggleWishlist = (propertyId: string) => {...};
  const isWishlisted = (propertyId: string) => wishlistedIds.includes(propertyId);
  const wishlistCount = wishlistedIds.length;

  return { wishlistedIds, toggleWishlist, isWishlisted, wishlistCount };
}
```
