```typescript
'use client';

import { useState } from 'react';
import { motion } from 'framer-motion';
import { toast } from 'sonner';
import {
  Gear,
  Buildings,
  Palette,
  Users,
  ShieldCheck,
  Plugs,
  CreditCard,
} from '@phosphor-icons/react';
import { cn } from '@/lib/utils';
import {
  ConfigPerfilAgencia,
  ConfigBranding,
  ConfigUsuarios,
  ConfigPermisos,
  ConfigIntegraciones,
  ConfigFacturacion,
} from '@/components/inmobiliaria';
import {
  MOCK_INMOBILIARIA_CONFIG_EXTENDED,
  MOCK_AGENCY_USERS,
  MOCK_INTEGRATIONS,
  MOCK_BILLING,
  MOCK_INVOICES,
} from '@/lib/data/mock-inmobiliaria';
import { DEFAULT_ROLE_PERMISSIONS } from '@/lib/types/inmobiliaria';

type ConfigTab = 'perfil' | 'branding' | 'usuarios' | 'permisos' | 'integraciones' | 'facturacion';

const TABS: { id: ConfigTab; label: string; icon: React.ElementType }[] = [
  { id: 'perfil', label: 'Perfil', icon: Buildings },
  { id: 'branding', label: 'Branding', icon: Palette },
  { id: 'usuarios', label: 'Usuarios', icon: Users },
  { id: 'permisos', label: 'Permisos', icon: ShieldCheck },
  { id: 'integraciones', label: 'Integraciones', icon: Plugs },
  { id: 'facturacion', label: 'Facturacion', icon: CreditCard },
];

export default function ConfiguracionPage() {
  const [activeTab, setActiveTab] = useState<ConfigTab>('perfil');
  const [config, setConfig] = useState(MOCK_INMOBILIARIA_CONFIG_EXTENDED);
  const [users, setUsers] = useState(MOCK_AGENCY_USERS);
  const [integrations, setIntegrations] = useState(MOCK_INTEGRATIONS);

  const handleSaveConfig = (newConfig: typeof config) => {
    setConfig(newConfig);
    toast.success('Configuracion guardada');
  };

  const handleSaveBranding = (branding: typeof config.branding) => {
    setConfig(prev => ({ ...prev, branding }));
    toast.success('Branding actualizado');
  };

  // ... handlers for other tabs

  return (
    <div className="p-4 md:p-6 space-y-6">
      {/* Header */}
      <div>
        <h1 className="text-2xl font-bold text-foreground flex items-center gap-2">
          <Gear className="w-7 h-7" />
          Configuracion
        </h1>
        <p className="text-muted-foreground mt-1">
          Administra la configuracion de tu inmobiliaria
        </p>
      </div>

      {/* Tabs */}
      <div className="flex flex-wrap gap-2 border-b border-border pb-2">
        {TABS.map(tab => (
          <button
            key={tab.id}
            onClick={() => setActiveTab(tab.id)}
            className={cn(
              'flex items-center gap-2 px-4 py-2 rounded-lg text-sm font-medium transition-colors',
              activeTab === tab.id
                ? 'bg-indigo-100 text-indigo-700 dark:bg-indigo-900/30 dark:text-indigo-400'
                : 'text-muted-foreground hover:bg-muted hover:text-foreground'
            )}
          >
            <tab.icon className="w-4 h-4" />
            {tab.label}
          </button>
        ))}
      </div>

      {/* Tab Content */}
      <motion.div
        key={activeTab}
        initial={{ opacity: 0, y: 10 }}
        animate={{ opacity: 1, y: 0 }}
        transition={{ duration: 0.2 }}
      >
        {activeTab === 'perfil' && (
          <ConfigPerfilAgencia config={config} onSave={handleSaveConfig} />
        )}
        {activeTab === 'branding' && (
          <ConfigBranding branding={config.branding} onSave={handleSaveBranding} />
        )}
        {activeTab === 'usuarios' && (
          <ConfigUsuarios users={users} onInvite={() => toast.info('Invitacion enviada')} />
        )}
        {activeTab === 'permisos' && (
          <ConfigPermisos permissions={DEFAULT_ROLE_PERMISSIONS} onSave={() => toast.success('Permisos actualizados')} />
        )}
        {activeTab === 'integraciones' && (
          <ConfigIntegraciones integrations={integrations} onToggle={(id, enabled) => {
            setIntegrations(prev => prev.map(i => i.id === id ? { ...i, isEnabled: enabled } : i));
            toast.success(enabled ? 'Integracion activada' : 'Integracion desactivada');
          }} />
        )}
        {activeTab === 'facturacion' && (
          <ConfigFacturacion billing={MOCK_BILLING} invoices={MOCK_INVOICES} />
        )}
      </motion.div>
    </div>
  );
}
```

```typescript
'use client';

import { useState } from 'react';
import { motion } from 'framer-motion';
import { toast } from 'sonner';
import {
  FileText,
  FolderOpen,
  FilePlus,
  ClipboardText,
} from '@phosphor-icons/react';
import { cn } from '@/lib/utils';
import {
  DocumentoTemplates,
  DocumentoManager,
  ActaEntregaForm,
  ActaEntregaViewer,
} from '@/components/inmobiliaria';
import {
  MOCK_DOCUMENT_TEMPLATES,
  MOCK_PROPERTY_DOCUMENTS,
  MOCK_ACTAS_ENTREGA,
  MOCK_CONSIGNACIONES,
} from '@/lib/data/mock-inmobiliaria';
import type { DocumentTemplate, PropertyDocument, ActaEntrega } from '@/lib/types/inmobiliaria';
import {
  Sheet,
  SheetContent,
} from '@/components/ui/sheet';

type DocTab = 'documentos' | 'plantillas' | 'actas';

export default function DocumentosPage() {
  const [activeTab, setActiveTab] = useState<DocTab>('documentos');
  const [selectedActa, setSelectedActa] = useState<ActaEntrega | null>(null);
  const [isActaFormOpen, setIsActaFormOpen] = useState(false);
  const [isActaViewerOpen, setIsActaViewerOpen] = useState(false);

  const handlePreviewTemplate = (template: DocumentTemplate) => {
    toast.info(`Vista previa: ${template.name}`);
  };

  const handleUseTemplate = (template: DocumentTemplate) => {
    toast.success(`Usando plantilla: ${template.name}`);
  };

  const handleViewDocument = (doc: PropertyDocument) => {
    toast.info(`Abriendo: ${doc.name}`);
  };

  const handleViewActa = (acta: ActaEntrega) => {
    setSelectedActa(acta);
    setIsActaViewerOpen(true);
  };

  return (
    <div className="p-4 md:p-6 space-y-6">
      {/* Header */}
      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
        <div>
          <h1 className="text-2xl font-bold text-foreground flex items-center gap-2">
            <FileText className="w-7 h-7" />
            Documentos
          </h1>
          <p className="text-muted-foreground mt-1">
            Gestiona documentos, plantillas y actas
          </p>
        </div>
        <button
          onClick={() => setIsActaFormOpen(true)}
          className="inline-flex items-center gap-2 px-4 py-2.5 rounded-xl bg-indigo-500 text-white hover:bg-indigo-600 font-medium shadow-lg shadow-indigo-500/25"
        >
          <ClipboardText className="w-5 h-5" />
          Nueva Acta de Entrega
        </button>
      </div>

      {/* Tabs */}
      <div className="flex gap-2 border-b border-border pb-2">
        <button
          onClick={() => setActiveTab('documentos')}
          className={cn(
            'flex items-center gap-2 px-4 py-2 rounded-lg text-sm font-medium transition-colors',
            activeTab === 'documentos'
              ? 'bg-indigo-100 text-indigo-700 dark:bg-indigo-900/30 dark:text-indigo-400'
              : 'text-muted-foreground hover:bg-muted'
          )}
        >
          <FolderOpen className="w-4 h-4" />
          Documentos
          <span className="text-xs bg-muted px-1.5 py-0.5 rounded-full">
            {MOCK_PROPERTY_DOCUMENTS.length}
          </span>
        </button>
        <button
          onClick={() => setActiveTab('plantillas')}
          className={cn(
            'flex items-center gap-2 px-4 py-2 rounded-lg text-sm font-medium transition-colors',
            activeTab === 'plantillas'
              ? 'bg-indigo-100 text-indigo-700 dark:bg-indigo-900/30 dark:text-indigo-400'
              : 'text-muted-foreground hover:bg-muted'
          )}
        >
          <FilePlus className="w-4 h-4" />
          Plantillas
        </button>
        <button
          onClick={() => setActiveTab('actas')}
          className={cn(
            'flex items-center gap-2 px-4 py-2 rounded-lg text-sm font-medium transition-colors',
            activeTab === 'actas'
              ? 'bg-indigo-100 text-indigo-700 dark:bg-indigo-900/30 dark:text-indigo-400'
              : 'text-muted-foreground hover:bg-muted'
          )}
        >
          <ClipboardText className="w-4 h-4" />
          Actas
          <span className="text-xs bg-muted px-1.5 py-0.5 rounded-full">
            {MOCK_ACTAS_ENTREGA.length}
          </span>
        </button>
      </div>

      {/* Content */}
      <motion.div
        key={activeTab}
        initial={{ opacity: 0, y: 10 }}
        animate={{ opacity: 1, y: 0 }}
      >
        {activeTab === 'documentos' && (
          <DocumentoManager
            documents={MOCK_PROPERTY_DOCUMENTS}
            onView={handleViewDocument}
          />
        )}
        {activeTab === 'plantillas' && (
          <DocumentoTemplates
            templates={MOCK_DOCUMENT_TEMPLATES}
            onPreview={handlePreviewTemplate}
            onUseTemplate={handleUseTemplate}
          />
        )}
        {activeTab === 'actas' && (
          <div className="grid gap-4">
            {MOCK_ACTAS_ENTREGA.map(acta => (
              <div
                key={acta.id}
                onClick={() => handleViewActa(acta)}
                className="p-4 rounded-xl border border-border bg-card hover:shadow-md transition-shadow cursor-pointer"
              >
                <div className="flex items-center justify-between">
                  <div>
                    <p className="font-medium text-foreground">{acta.propertyTitle}</p>
                    <p className="text-sm text-muted-foreground">{acta.tenantName}</p>
                  </div>
                  <span className={cn(
                    'px-2 py-1 rounded-full text-xs font-medium',
                    acta.status === 'completed' ? 'bg-emerald-100 text-emerald-700' : 'bg-amber-100 text-amber-700'
                  )}>
                    {acta.status}
                  </span>
                </div>
              </div>
            ))}
          </div>
        )}
      </motion.div>

      {/* Acta Form Sheet */}
      <Sheet open={isActaFormOpen} onOpenChange={setIsActaFormOpen}>
        <SheetContent className="w-full sm:max-w-2xl overflow-y-auto">
          <ActaEntregaForm
            consignaciones={MOCK_CONSIGNACIONES}
            onSave={() => {
              setIsActaFormOpen(false);
              toast.success('Acta creada');
            }}
            onCancel={() => setIsActaFormOpen(false)}
          />
        </SheetContent>
      </Sheet>

      {/* Acta Viewer Sheet */}
      <Sheet open={isActaViewerOpen} onOpenChange={setIsActaViewerOpen}>
        <SheetContent className="w-full sm:max-w-2xl overflow-y-auto">
          {selectedActa && <ActaEntregaViewer acta={selectedActa} />}
        </SheetContent>
      </Sheet>
    </div>
  );
}
```

```typescript
'use client';

import { useState } from 'react';
import { motion } from 'framer-motion';
import { toast } from 'sonner';
import {
  ChartLineUp,
  ChartBar,
  TrendUp,
  ArrowsClockwise,
  Export,
} from '@phosphor-icons/react';
import { cn } from '@/lib/utils';
import {
  AnalyticsDashboard,
  AnalyticsKPICards,
  AnalyticsTrends,
  AnalyticsForecasting,
} from '@/components/inmobiliaria';
import {
  MOCK_ANALYTICS_DATA,
  MOCK_TREND_ANALYSIS,
  MOCK_FORECAST_DATA,
} from '@/lib/data/mock-inmobiliaria';

type AnalyticsView = 'dashboard' | 'trends' | 'forecasting';

export default function AnalyticsPage() {
  const [activeView, setActiveView] = useState<AnalyticsView>('dashboard');
  const [isRefreshing, setIsRefreshing] = useState(false);

  const handleRefresh = async () => {
    setIsRefreshing(true);
    await new Promise(resolve => setTimeout(resolve, 1500));
    setIsRefreshing(false);
    toast.success('Datos actualizados');
  };

  const handleExport = (format: 'pdf' | 'excel') => {
    toast.info(`Exportando a ${format.toUpperCase()}...`);
  };

  return (
    <div className="p-4 md:p-6 space-y-6">
      {/* Header */}
      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
        <div>
          <h1 className="text-2xl font-bold text-foreground flex items-center gap-2">
            <ChartLineUp className="w-7 h-7" />
            Analitica
          </h1>
          <p className="text-muted-foreground mt-1">
            Metricas avanzadas, tendencias y proyecciones
          </p>
        </div>
        <div className="flex items-center gap-3">
          <button
            onClick={handleRefresh}
            disabled={isRefreshing}
            className={cn(
              'inline-flex items-center gap-2 px-4 py-2.5 rounded-xl border border-border bg-card text-muted-foreground hover:text-foreground transition-colors',
              isRefreshing && 'opacity-50 cursor-not-allowed'
            )}
          >
            <ArrowsClockwise className={cn('w-5 h-5', isRefreshing && 'animate-spin')} />
            Actualizar
          </button>
          <button
            onClick={() => handleExport('excel')}
            className="inline-flex items-center gap-2 px-4 py-2.5 rounded-xl bg-indigo-500 text-white hover:bg-indigo-600 font-medium shadow-lg shadow-indigo-500/25"
          >
            <Export className="w-5 h-5" />
            Exportar
          </button>
        </div>
      </div>

      {/* View Tabs */}
      <div className="flex gap-2 border-b border-border pb-2">
        <button
          onClick={() => setActiveView('dashboard')}
          className={cn(
            'flex items-center gap-2 px-4 py-2 rounded-lg text-sm font-medium transition-colors',
            activeView === 'dashboard'
              ? 'bg-indigo-100 text-indigo-700 dark:bg-indigo-900/30 dark:text-indigo-400'
              : 'text-muted-foreground hover:bg-muted'
          )}
        >
          <ChartBar className="w-4 h-4" />
          Dashboard
        </button>
        <button
          onClick={() => setActiveView('trends')}
          className={cn(
            'flex items-center gap-2 px-4 py-2 rounded-lg text-sm font-medium transition-colors',
            activeView === 'trends'
              ? 'bg-indigo-100 text-indigo-700 dark:bg-indigo-900/30 dark:text-indigo-400'
              : 'text-muted-foreground hover:bg-muted'
          )}
        >
          <ChartLineUp className="w-4 h-4" />
          Tendencias
        </button>
        <button
          onClick={() => setActiveView('forecasting')}
          className={cn(
            'flex items-center gap-2 px-4 py-2 rounded-lg text-sm font-medium transition-colors',
            activeView === 'forecasting'
              ? 'bg-indigo-100 text-indigo-700 dark:bg-indigo-900/30 dark:text-indigo-400'
              : 'text-muted-foreground hover:bg-muted'
          )}
        >
          <TrendUp className="w-4 h-4" />
          Proyecciones
        </button>
      </div>

      {/* Content */}
      <motion.div
        key={activeView}
        initial={{ opacity: 0, y: 10 }}
        animate={{ opacity: 1, y: 0 }}
      >
        {activeView === 'dashboard' && (
          <div className="space-y-6">
            <AnalyticsKPICards kpis={MOCK_ANALYTICS_DATA.kpis} />
            <AnalyticsDashboard
              data={MOCK_ANALYTICS_DATA}
              onExport={handleExport}
              isLoading={isRefreshing}
            />
          </div>
        )}
        {activeView === 'trends' && (
          <AnalyticsTrends data={MOCK_TREND_ANALYSIS} />
        )}
        {activeView === 'forecasting' && (
          <AnalyticsForecasting data={MOCK_FORECAST_DATA} />
        )}
      </motion.div>
    </div>
  );
}
```
