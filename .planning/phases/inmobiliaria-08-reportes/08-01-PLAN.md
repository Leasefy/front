// Helper to generate mock report data
export function generateComisionesAgenteReport(
  period: { start: string; end: string }
): ComisionesAgenteReport {
  const items = MOCK_AGENTES.map((agente) => ({
    agenteId: agente.id,
    agenteName: agente.name,
    closedDeals: agente.closedThisMonth,
    totalCommission: agente.commissionsThisMonth,
    avgCommission: agente.closedThisMonth > 0
      ? agente.commissionsThisMonth / agente.closedThisMonth
      : 0,
    topProperty: agente.properties[0] || undefined,
  }));

  return {
    generatedAt: new Date().toISOString(),
    period,
    items,
    summary: {
      totalAgents: items.length,
      totalCommissions: items.reduce((sum, i) => sum + i.totalCommission, 0),
      avgPerAgent: items.reduce((sum, i) => sum + i.totalCommission, 0) / items.length,
    },
  };
}

export function generateOcupacionReport(): OcupacionReport {
  const zones = ['Norte', 'Sur', 'Centro', 'Occidente'];
  const zoneData = zones.map((zone) => {
    const total = Math.floor(Math.random() * 20) + 10;
    const occupied = Math.floor(total * (0.7 + Math.random() * 0.25));
    const available = total - occupied - Math.floor(Math.random() * 3);
    const inProcess = total - occupied - available;
    return {
      zone,
      total,
      occupied,
      available,
      inProcess,
      occupancyRate: Math.round((occupied / total) * 100),
    };
  });

  const overall = {
    total: zoneData.reduce((sum, z) => sum + z.total, 0),
    occupied: zoneData.reduce((sum, z) => sum + z.occupied, 0),
    occupancyRate: 0,
  };
  overall.occupancyRate = Math.round((overall.occupied / overall.total) * 100);

  return {
    generatedAt: new Date().toISOString(),
    zones: zoneData,
    overall,
  };
}

export function generateVencimientosReport(): VencimientosReport {
  const items = MOCK_CONSIGNACIONES
    .filter((c) => c.status === 'rented' && c.currentLease)
    .map((c) => {
      const leaseEnd = c.currentLease!.endDate;
      const daysUntilExpiry = Math.floor(
        (new Date(leaseEnd).getTime() - Date.now()) / (1000 * 60 * 60 * 24)
      );
      return {
        propertyId: c.propertyId,
        propertyTitle: c.propertyTitle,
        tenantName: c.currentLease!.tenantName,
        propietarioName: c.propietarioName,
        leaseEnd,
        daysUntilExpiry,
        renewalStatus: daysUntilExpiry < 30 ? 'pending' : 'pending' as const,
      };
    })
    .filter((item) => item.daysUntilExpiry <= 90);

  return {
    generatedAt: new Date().toISOString(),
    items,
    summary: {
      expiring30: items.filter((i) => i.daysUntilExpiry <= 30).length,
      expiring60: items.filter((i) => i.daysUntilExpiry > 30 && i.daysUntilExpiry <= 60).length,
      expiring90: items.filter((i) => i.daysUntilExpiry > 60 && i.daysUntilExpiry <= 90).length,
      total: items.length,
    },
  };
}

export function generateFlujoCajaReport(
  period: { start: string; end: string }
): FlujoCajaReport {
  const months = ['Ene', 'Feb', 'Mar', 'Abr', 'May', 'Jun'];
  const monthData = months.map((month) => {
    const ingresos = Math.floor(Math.random() * 50000000) + 80000000;
    const comisiones = Math.floor(ingresos * 0.08);
    const dispersiones = ingresos - comisiones;
    return {
      month,
      ingresos,
      dispersiones,
      comisiones,
      balance: comisiones,
    };
  });

  return {
    generatedAt: new Date().toISOString(),
    period,
    months: monthData,
    summary: {
      totalIngresos: monthData.reduce((sum, m) => sum + m.ingresos, 0),
      totalDispersiones: monthData.reduce((sum, m) => sum + m.dispersiones, 0),
      totalComisiones: monthData.reduce((sum, m) => sum + m.comisiones, 0),
      netBalance: monthData.reduce((sum, m) => sum + m.balance, 0),
    },
  };
}
